import requests
import json

OLLAMA_URL = "http://localhost:11434/api/chat"
DEFAULT_MODEL = "qwen2.5-coder:3b"

def ask_llm(messages, model=None):
    use_model = model or DEFAULT_MODEL
    response = requests.post(
        OLLAMA_URL,
        json={"model": use_model, "messages": messages, "stream": False},
        timeout=120
    )
    response.raise_for_status()
    return response.json()["message"]["content"]

if __name__ == "__main__":
    print(ask_llm([{"role": "user", "content": "Привет"}]))

$code = @'
"""Image Generator Agent — generiruet izobrazheniya cherez Pollinations.ai (besplatno, bez klyucha)."""
import sys
import requests
import urllib.parse
from pathlib import Path

sys.path.insert(0, str(Path(__file__).parent.parent / "Config"))
from llm_client import ask_llm

IMAGES_DIR = "images"

def generate_image(prompt: str, output_path: Path, width=1024, height=1024):
    """Generiruet izobrazhenie po promptu i sohranyaet."""
    encoded = urllib.parse.quote(prompt)
    url = f"https://image.pollinations.ai/prompt/{encoded}?width={width}&height={height}&nologo=true&seed=42"
    print(f"   Generatsiya: {prompt[:50]}...")
    response = requests.get(url, timeout=120)
    response.raise_for_status()
    output_path.write_bytes(response.content)
    print(f"   Sohraneno: {output_path} ({len(response.content)//1024} KB)")
    return str(output_path)

def run_image_generator(image_prompts: list, project_dir: str):
    """Generiruet vse izobrazheniya dlya proekta."""
    project_path = Path(project_dir)
    img_dir = project_path / IMAGES_DIR
    img_dir.mkdir(parents=True, exist_ok=True)

    paths = []
    for i, prompt in enumerate(image_prompts, 1):
        out = img_dir / f"img_{i}.jpg"
        try:
            p = generate_image(prompt, out)
            paths.append(p)
        except Exception as e:
            print(f"   [!] Oshibka generatsii {i}: {e}")
    return paths

if __name__ == "__main__":
    if len(sys.argv) < 3:
        print('Ispolzovanie: python Agents/image_generator.py \'["prompt1","prompt2"]\' ./Projects/name')
        sys.exit(1)
    import json
    prompts = json.loads(sys.argv[1])
    run_image_generator(prompts, sys.argv[2])
'@

def structure_existing_tz(text: str):
    prompt = f"""Ty - Brief Agent. U tebya uzhe est gotovoe TZ ot klienta.
Pereformatiruy ego v chistyy Markdown s zagolovkami.
Dobav razdel ## Image Prompts s 3-5 promptami na angliyskom dlya AI-foto.

ISKHODNYY TEKST:
{text}

Rezultat:"""
    messages = [{"role": "user", "content": prompt}]
    return ask_llm(messages, model="qwen2.5-coder:3b")

def extract_image_prompts(tz_text: str):
    """Izvlekaet prompty iz Markdown."""
    prompts = []
    lines = tz_text.split("\n")
    in_prompts = False
    for line in lines:
        if "## image prompts" in line.lower() or "**image prompts**" in line.lower():
            in_prompts = True
            continue
        if in_prompts and line.startswith("##"):
            break
        if in_prompts and line.strip().startswith("-") or line.strip().startswith("*"):
            p = line.strip().lstrip("-* ").strip()
            if p and len(p) > 10:
                prompts.append(p)
    return prompts

def run_brief(task: str, project_dir: str):
    project_path = Path(project_dir)
    project_path.mkdir(parents=True, exist_ok=True)

    brief_file = project_path / "brief.md"
    prompts_file = project_path / "image_prompts.json"

    if len(task) > 300:
        print(f"Brief Agent: obnaruzheno gotovoe TZ ({len(task)} simvolov)")
        tz = structure_existing_tz(task)
    else:
        print(f"Brief Agent: sostavlyayu TZ dlya '{task[:50]}...'")
        messages = [
            {"role": "system", "content": SYSTEM_PROMPT},
            {"role": "user", "content": f"Zapros: {task}"}
        ]
        tz = ask_llm(messages, model="qwen2.5-coder:7b")

    brief_file.write_text(tz, encoding="utf-8")
    prompts = extract_image_prompts(tz)
    prompts_file.write_text(json.dumps(prompts, ensure_ascii=False, indent=2), encoding="utf-8")

    print(f"   TZ sohraneno: {brief_file}")
    print(f"   Promptov dlya izobrazheniy: {len(prompts)}")
    return str(brief_file)

if __name__ == "__main__":
    if len(sys.argv) < 3:
        print('Ispolzovanie: python Agents/brief.py "zadacha" ./put_k_proektu')
        sys.exit(1)
    task_arg = sys.argv[1]
    project_dir_arg = sys.argv[2]
    run_brief(task_arg, project_dir_arg)

"""Image Generator Agent — multi-provider: Unsplash (stock), FLUX (AI), Pollinations (fallback)."""
import sys
import requests
import urllib.parse
import random
from pathlib import Path

# ======== VSTAV SVOI KLYUCHI ========
UNSPLASH_ACCESS_KEY = ""   # https://unsplash.com/developers
HUGGINGFACE_TOKEN = ""     # https://huggingface.co/settings/tokens

UNSPLASH_API = "https://api.unsplash.com/search/photos"
HF_API = "https://api-inference.huggingface.co/models/black-forest-labs/FLUX.1-schnell"
POLLINATIONS_URL = "https://image.pollinations.ai/prompt/{}?width=1024&height=1024&nologo=true&seed={}"

IMAGES_DIR = "images"


def download_image(url: str, output_path: Path):
    r = requests.get(url, timeout=60, stream=True)
    r.raise_for_status()
    output_path.write_bytes(r.content)
    return str(output_path)


def unsplash_search(query: str, output_path: Path):
    """Kachestvennye REALNYE foto. Trebuet klyuch."""
    if not UNSPLASH_ACCESS_KEY:
        raise ValueError("Unsplash key ne ustanovlen")
    headers = {"Authorization": f"Client-ID {UNSPLASH_ACCESS_KEY}"}
    params = {"query": query, "per_page": 5, "orientation": "landscape"}
    r = requests.get(UNSPLASH_API, headers=headers, params=params, timeout=30)
    r.raise_for_status()
    data = r.json()
    if not data["results"]:
        raise ValueError("Nichto ne naydeno")
    photo = random.choice(data["results"])
    img_url = photo["urls"]["regular"]  # 1080px
    print(f"   [Unsplash] Foto ot {photo['user']['name']}")
    return download_image(img_url, output_path)


def flux_generate(prompt: str, output_path: Path):
    """AI-generatsiya FLUX.1-schnell. Vysokoye kachestvo."""
    if not HUGGINGFACE_TOKEN:
        raise ValueError("HF token ne ustanovlen")
    headers = {"Authorization": f"Bearer {HUGGINGFACE_TOKEN}"}
    payload = {"inputs": prompt}
    r = requests.post(HF_API, headers=headers, json=payload, timeout=120)
    r.raise_for_status()
    output_path.write_bytes(r.content)
    print(f"   [FLUX] Generatsiya: {output_path.name}")
    return str(output_path)


def pollinations_generate(prompt: str, output_path: Path):
    """Fallback. Bez klyucha."""
    encoded = urllib.parse.quote(prompt)
    url = POLLINATIONS_URL.format(encoded, random.randint(1, 9999))
    print(f"   [Pollinations] Fallback...")
    return download_image(url, output_path)


def generate_image(prompt: str, output_path: Path, mode="auto"):
    """
    mode:
        "stock"  -> Unsplash (realnye foto)
        "ai"     -> FLUX (generatsiya)
        "auto"   -> Unsplash -> FLUX -> Pollinations
    """
    if mode == "stock" and UNSPLASH_ACCESS_KEY:
        try:
            return unsplash_search(prompt, output_path)
        except Exception as e:
            print(f"   [!] Unsplash: {e}")

    if mode == "ai" and HUGGINGFACE_TOKEN:
        try:
            return flux_generate(prompt, output_path)
        except Exception as e:
            print(f"   [!] FLUX: {e}")

    # Auto-prioritet
    if UNSPLASH_ACCESS_KEY:
        try:
            return unsplash_search(prompt, output_path)
        except Exception:
            pass

    if HUGGINGFACE_TOKEN:
        try:
            return flux_generate(prompt, output_path)
        except Exception:
            pass

    return pollinations_generate(prompt, output_path)


def run_image_generator(image_tasks: list, project_dir: str):
    project_path = Path(project_dir)
    img_dir = project_path / IMAGES_DIR
    img_dir.mkdir(parents=True, exist_ok=True)

    paths = []
    for task in image_tasks:
        prompt = task.get("prompt", "abstract")
        mode = task.get("mode", "auto")
        fname = task.get("filename", f"img_{len(paths)+1}.jpg")
        out = img_dir / fname
        try:
            p = generate_image(prompt, out, mode)
            paths.append(p)
        except Exception as e:
            print(f"   [!] Oshibka {fname}: {e}")
    return paths


if __name__ == "__main__":
    if len(sys.argv) < 3:
        print('Ispolzovanie: python Agents/image_generator.py \'[{"prompt":"coffee","mode":"stock"}]\' ./Projects/name')
        sys.exit(1)
    import json
    tasks = json.loads(sys.argv[1])
    run_image_generator(tasks, sys.argv[2])

def extract_image_tasks(tz_text: str):
    tasks = []
    lines = tz_text.split("\n")
    in_images = False
    for line in lines:
        if "## image" in line.lower():
            in_images = True
            continue
        if in_images and line.startswith("##"):
            break
        if in_images and (line.strip().startswith("-") or line.strip().startswith("*")):
            text = line.strip().lstrip("-* ").strip()
            if text.startswith("[stock]"):
                mode, prompt = "stock", text[7:].strip()
            elif text.startswith("[ai]"):
                mode, prompt = "ai", text[4:].strip()
            else:
                mode, prompt = "auto", text
            if prompt:
                tasks.append({"prompt": prompt, "mode": mode, "filename": f"img_{len(tasks)+1}.jpg"})
    return tasks

def api_get(url: str, params=None):
    import requests
    headers = {}
    if GITHUB_TOKEN:
        headers["Authorization"] = f"token {GITHUB_TOKEN}"
    try:
        r = requests.get(url, headers=headers, params=params, timeout=30)
        if r.status_code == 403:
            print("   [!] GitHub API limit. Zhdem 60 sek...")
            time.sleep(60)
            return None
        if r.status_code == 422:
            print("   [!] Nevalidnyy zapros")
            return None
        r.raise_for_status()
        return r.json()
    except Exception as e:
        print(f"   [!] API error: {e}")
        return None

def search_repos(query: str, per_page=5):
    params = {"q": query, "sort": "stars", "order": "desc", "per_page": per_page}
    data = api_get(GITHUB_API, params)
    return data.get("items", []) if data else []

def clone_repo(repo_url: str, dest: Path):
    """Kloniruet repo vo vremennuyu papku."""
    if dest.exists():
        shutil.rmtree(dest)
    result = subprocess.run(
        ["git", "clone", "--depth", "1", repo_url, str(dest)],
        capture_output=True, text=True
    )
    return result.returncode == 0

def find_entry_file(repo_path: Path):
    """Ishchet glavnyy HTML-fayl."""
    candidates = [
        repo_path / "index.html",
        repo_path / "dist" / "index.html",
        repo_path / "build" / "index.html",
        repo_path / "public" / "index.html",
        repo_path / "src" / "index.html",
    ]
    for c in candidates:
        if c.exists():
            return c
    # Ishchem lyuboy index.html
    for html in repo_path.rglob("index.html"):
        return html
    return None

def has_node_modules_refs(repo_path: Path):
    """Proveryayet, est li ssylki na node_modules v HTML."""
    entry = find_entry_file(repo_path)
    if not entry:
        return False, []
    content = entry.read_text(encoding="utf-8", errors="ignore")
    issues = []
    if "node_modules" in content:
        issues.append("node_modules v HTML")
    if "../" in content and "vendor" in content.lower():
        issues.append("vendor paths")
    # Proveryayem CSS/JS fayly
    for link in re.findall(r'href=["\'](.*?)["\']', content):
        if link.startswith("node_modules") or link.startswith("../node_modules"):
            issues.append(f"link: {link}")
    for src in re.findall(r'src=["\'](.*?)["\']', content):
        if src.startswith("node_modules") or src.startswith("../node_modules"):
            issues.append(f"src: {src}")
    return len(issues) > 0, issues

def analyze_with_llm(repo_path: Path, repo_name: str, repo_desc: str):
    """Ispolzuyet LLM dlya sozdaniya skill.json."""
    entry = find_entry_file(repo_path)
    if not entry:
        return None
    
    # Chitaem pervyye 80 strok HTML dlya analiza
    html_preview = entry.read_text(encoding="utf-8", errors="ignore")[:3000]
    
    # Smotrim strukturu
    structure = []
    for item in repo_path.iterdir():
        if item.is_dir():
            structure.append(f"📁 {item.name}/")
        else:
            structure.append(f"📄 {item.name}")
    
    prompt = f"""Ty - arkhitektor veb-shablonov. Proanaliziruy etot repozitoriy i sozday metadata dlya skill.json.

NAZVANIYE REPO: {repo_name}
OPISANIYE: {repo_desc}

STRUKTURA:
{chr(10).join(structure[:15])}

PERVYYE STROKI HTML:
{html_preview[:1500]}

Zadacha: verni JSON:
{{
  "name": "korotkoye_imya_na_angliyskom",
  "description": "1-2 predlozheniya o chem etot shablon",
  "entry": "put k glavnomu HTML (index.html ili dist/index.html)",
  "category": "landing | portfolio | ecommerce | business | other",
  "has_dist": true/false,
  "needs_build": true/false,
  "quality_score": 1-10
}}

Pravila:
- entry: otnositelnyy put vnutri repo (bez imeni repo)
- has_dist: true esli est papka dist/ ili build/
- needs_build: true esli est package.json i net dist/
- quality_score: 10 = idealnyy static HTML, 1 = nuzhna sborka/webpack
- Ne pishi kod. Tolko JSON."""'

    messages = [{"role": "user", "content": prompt}]
    try:
        answer = ask_llm(messages, model="qwen2.5-coder:7b")
        # Chistim JSON
        cleaned = answer.strip()
        if cleaned.startswith("```json"):
            cleaned = cleaned[7:]
        if cleaned.startswith("```"):
            cleaned = cleaned[3:]
        if cleaned.endswith("```"):
            cleaned = cleaned[:-3]
        cleaned = cleaned.strip()
        meta = json.loads(cleaned)
        return meta
    except Exception as e:
        print(f"   [!] LLM ne spravilsya: {e}")
        return None

def fix_paths(repo_path: Path):
    """Zamenyayet lokalnye vendor paths na CDN gde vozmozhno."""
    entry = find_entry_file(repo_path)
    if not entry:
        return
    content = entry.read_text(encoding="utf-8")
    
    # Zamenyayem bootstrap na CDN
    content = re.sub(
        r'href=["\'].*?bootstrap.*?\.css["\']',
        'href="https://cdn.jsdelivr.net/npm/bootstrap@5.3.2/dist/css/bootstrap.min.css"',
        content
    )
    content = re.sub(
        r'src=["\'].*?bootstrap.*?\.js["\']',
        'src="https://cdn.jsdelivr.net/npm/bootstrap@5.3.2/dist/js/bootstrap.bundle.min.js"',
        content
    )
    # Zamenyayem jquery
    content = re.sub(
        r'src=["\'].*?jquery.*?\.js["\']',
        'src="https://code.jquery.com/jquery-3.7.1.min.js"',
        content
    )
    
    entry.write_text(content, encoding="utf-8")

def install_skill_from_repo(repo_url: str, repo_name: str, repo_desc: str, temp_dir: Path):
    """Ustanavlivayet skill iz uje sklonirovannogo repo."""
    target = SKILLS_DIR / repo_name
    if target.exists():
        print(f"   [!] Skill '{repo_name}' uzhe est. Propuskayu.")
        return False, "already_exists"
    
    meta = analyze_with_llm(temp_dir, repo_name, repo_desc)
    if not meta:
        return False, "llm_failed"
    
    # Esli nuzhna sborka — propuskayem (poka ne umeem)
    if meta.get("needs_build", False) and not meta.get("has_dist", False):
        print(f"   [!] Trebuetsya sborka (npm/webpack). Propuskayu.")
        return False, "needs_build"
    
    # Kopiruyem
    target.mkdir(parents=True, exist_ok=True)
    
    # Opredelyayem chto kopirovat
    entry_rel = meta.get("entry", "index.html")
    entry_path = temp_dir / entry_rel
    
    if entry_path.exists() and "dist" in entry_rel or "build" in entry_rel:
        # Kopiruyem soderzhimoye dist/ v koren
        src_dir = entry_path.parent
        for item in src_dir.rglob("*"):
            if item.is_file():
                rel = item.relative_to(src_dir)
                dest = target / rel
                dest.parent.mkdir(parents=True, exist_ok=True)
                shutil.copy2(item, dest)
        meta["entry"] = "index.html"  # Teper v korne
    else:
        # Kopiruyem vse repo
        for item in temp_dir.iterdir():
            if item.name == ".git":
                continue
            dest = target / item.name
            if item.is_dir():
                shutil.copytree(item, dest, dirs_exist_ok=True)
            else:
                shutil.copy2(item, dest)
    
    # Fix paths
    fix_paths(target)
    
    # Sozdayem skill.json
    skill_json = {
        "name": meta.get("name", repo_name),
        "version": "1.0.0",
        "description": meta.get("description", f"Auto-installed from {repo_url}"),
        "entry": meta.get("entry", "index.html"),
        "category": meta.get("category", "other"),
        "source": repo_url,
        "auto_installed": True,
        "dependencies": []
    }
    (target / "skill.json").write_text(json.dumps(skill_json, indent=2, ensure_ascii=False), encoding="utf-8")
    
    return True, "success"

def log_result(repo_name: str, repo_url: str, status: str, reason: str):
    """Sokhranyayet log."""
    LOG_FILE.parent.mkdir(parents=True, exist_ok=True)
    logs = []
    if LOG_FILE.exists():
        logs = json.loads(LOG_FILE.read_text(encoding="utf-8"))
    logs.append({
        "timestamp": datetime.now().isoformat(),
        "repo": repo_name,
        "url": repo_url,
        "status": status,
        "reason": reason
    })
    LOG_FILE.write_text(json.dumps(logs, indent=2, ensure_ascii=False), encoding="utf-8")

def run_hunt(max_per_query=3, max_total=10):
    """Glavnaya funktsiya."""
    print("=" * 60)
    print("AUTOSKILL HUNTER ZAPUSHCHEN")
    print("=" * 60)
    
    found = []
    installed = 0
    skipped = 0
    
    for query in SEARCH_QUERIES:
        if installed >= max_total:
            break
        print(f"\n[*] Poisk: '{query}'")
        repos = search_repos(query, per_page=max_per_query + 2)
        
        for repo in repos:
            if installed >= max_total:
                break
            
            repo_name = repo.get("name", "unknown")
            repo_url = repo.get("clone_url", "")
            repo_desc = repo.get("description", "") or ""
            stars = repo.get("stargazers_count", 0)
            
            # Filtruyem: ne ustanavlivayem esli uzhe est
            if (SKILLS_DIR / repo_name).exists():
                continue
            
            print(f"\n[>] {repo_name} ({stars}⭐)")
            print(f"    {repo_url}")
            
            # Kloniruem vo vremennuyu papku
            with tempfile.TemporaryDirectory() as tmp:
                tmp_path = Path(tmp) / "repo"
                if not clone_repo(repo_url, tmp_path):
                    print("    [!] Ne udalos klonirovat")
                    log_result(repo_name, repo_url, "fail", "clone_error")
                    skipped += 1
                    continue
                
                # Proveryayem node_modules
                has_issues, issues = has_node_modules_refs(tmp_path)
                if has_issues:
                    print(f"    [!] Naideny problemy: {', '.join(issues[:3])}")
                    # Vse ravno probuyem — mozhet fix_paths pomozhet
                
                # Ustanavlivayem
                success, reason = install_skill_from_repo(repo_url, repo_name, repo_desc, tmp_path)
                if success:
                    print(f"    [+] Skill '{repo_name}' ustanovlen!")
                    installed += 1
                    log_result(repo_name, repo_url, "success", reason)
                else:
                    print(f"    [-] Propushen: {reason}")
                    skipped += 1
                    log_result(repo_name, repo_url, "skipped", reason)
            
            time.sleep(2)  # Ne nagruzim GitHub
    
    print("\n" + "=" * 60)
    print(f"GOTOV! Ustanovleno: {installed}, Propusheno: {skipped}")
    print(f"Log: {LOG_FILE}")
    print("=" * 60)

if __name__ == "__main__":
    import argparse
    parser = argparse.ArgumentParser()
    parser.add_argument("--max", type=int, default=10, help="Maksimum skillov dlya ustanovki")
    parser.add_argument("--per-query", type=int, default=3, help="Skolko repo prosmatrivat na zapros")
    args = parser.parse_args()
    
    run_hunt(max_per_query=args.per_query, max_total=args.max)

**Atlas — это не генератор лендингов. Это оркестратор цифровых продуктов.**

| Тип продукта | Что создаёт | Технологии |
|---|---|---|
| **Web** | Сайты, веб-приложения, админки | HTML/React/Next.js |
| **Games** | Браузерные игры, мини-игры, кликеры | Three.js, Phaser, Pygame |
| **Apps** | Python-утилиты, десктопные приложения, автоматизация | Python, Tkinter, PyQt |
| **Branding** | Логотипы (SVG), brand book, цветовые схемы, типографика | HTML/CSS/SVG |
| **AI Media** | Фотосессии, аватары, продуктовые фото, рекламные креативы | FLUX, Pollinations, Pillow |
| **Full Stack** | Backend + Frontend + DB | Python/FastAPI + React + SQLite |

---

## Что меняется в архитектуре

### 1. Product Router Agent
Новый агент **перед** Executive. Он читает запрос и решает: это сайт, игра, брендинг или фотосессия?

### 2. Skills → категории
Skills/ ├── web/ │ ├── modern_landing/ │ ├── agency/ │ ├── motion_premium/ │ └── product_showcase/ ├── games/ │ ├── threejs_starter/ │ └── phaser_platformer/ ├── apps/ │ ├── python_gui/ │ └── automation_script/ ├── branding/ │ └── brand_kit/ └── media/ └── photo_session/

### 3. Специализированные агенты
- `web_developer.py` — то, что сейчас `developer.py`
- `game_developer.py` — Three.js/Phaser игры
- `app_developer.py` — Python приложения
- `branding_agent.py` — логотипы, цвета, brand book
- `media_agent.py` — AI фотосессии, альбомы

"""Product Router — opredelyayet tip produkta iz zaprosa i vybyirayet agentov."""
import sys
import json
from pathlib import Path

sys.path.insert(0, str(Path(__file__).parent.parent / "Config"))
from llm_client import ask_llm

SYSTEM_PROMPT = """Ty - Product Router v Atlas Digital Studio.
Tvoya zadacha: iz zaprosa klienta opredelit TIP produkta i komandu.

Tipy produktov:
- web: sayty, lendingy, internet-magaziny, veb-prilozheniya
- game: igry, brauzernyye igry, mini-igry, klikery, platformery
- app: desktopnye prilozheniya, utilitiy, skripty avtomatizatsii, boty
- branding: logotipy, firovyy stil, brand book, tsvetovyye skhemy, tipografika
- media: ai-fotosessii, avatary, produktovyye foto, reklamnyye kreativy, albomy
- fullstack: backend + frontend + baza dannykh (prilozheniya s serverom)

Otvet STROGO v formate JSON:
{
  "product_type": "web|game|app|branding|media|fullstack",
  "confidence": 0-100,
  "description": "chto nuzhno sdelat",
  "agents": ["spisok agentov"],
  "skills": ["spisok skillov"],
  "notes": "dopolnitelnyye zametki"
}

Pravila:
- Esli zapros pro "sayt", "lending", "stranitsa" -> web
- Esli zapros pro "igru", "game", "platformer", "kliker" -> game
- Esli zapros pro "programmu", "prilozheniye", "utilita", "bot" -> app
- Esli zapros pro "logotip", "brend", "stil", "firmennyy" -> branding
- Esli zapros pro "foto", "fotosessiya", "avatar", "kreativ" -> media
- Esli zapros pro "sistem", "backend", "baza", "api" -> fullstack
- Confidence > 80 = tochno, < 50 = nuzhno utochnit"""

def route_product(task: str):
    messages = [
        {"role": "system", "content": SYSTEM_PROMPT},
        {"role": "user", "content": f"Zapros klienta: {task}"}
    ]
    print("Product Router analiziruyet...")
    answer = ask_llm(messages, model="qwen2.5-coder:7b")
    
    try:
        cleaned = answer.strip()
        if cleaned.startswith("```json"): cleaned = cleaned[7:]
        if cleaned.endswith("```"): cleaned = cleaned[:-3]
        cleaned = cleaned.strip()
        result = json.loads(cleaned)
        
        print(f"\nTip produkta: {result.get('product_type', 'unknown').upper()}")
        print(f"Uverennost: {result.get('confidence', 0)}%")
        print(f"Agenty: {', '.join(result.get('agents', []))}")
        print(f"Skills: {', '.join(result.get('skills', []))}")
        if result.get('notes'):
            print(f"Zametki: {result['notes']}")
        return result
    except Exception as e:
        print(f"[!] Oshibka marshrutizatsii: {e}")
        return {"product_type": "web", "agents": ["web_developer"], "skills": ["modern_landing"]}

if __name__ == "__main__":
    if len(sys.argv) < 2:
        print('Ispolzovaniye: python Agents/product_router.py "sozday brend dlya kofeyni"')
        sys.exit(1)
    route_product(sys.argv[1])

"""Media Agent - AI fotosessiya, avatary, produktovyye foto, albomy."""
import sys
import json
import requests
import urllib.parse
import random
from pathlib import Path

sys.path.insert(0, str(Path(__file__).parent.parent / "Config"))
from llm_client import ask_llm

POLLINATIONS_URL = "https://image.pollinations.ai/prompt/{}?width=1024&height=1024&nologo=true&seed={}"

def generate_image(prompt: str, output_path: Path, width=1024, height=1024):
    encoded = urllib.parse.quote(prompt)
    url = f"https://image.pollinations.ai/prompt/{encoded}?width={width}&height={height}&nologo=true&seed={random.randint(1,9999)}"
    print(f"   Generatsiya: {prompt[:50]}...")
    r = requests.get(url, timeout=120)
    r.raise_for_status()
    output_path.write_bytes(r.content)
    print(f"   OK: {len(r.content)//1024} KB")
    return str(output_path)

def create_photo_session(task: str, project_dir: str, count=5):
    project_path = Path(project_dir)
    project_path.mkdir(parents=True, exist_ok=True)
    img_dir = project_path / "photos"
    img_dir.mkdir(exist_ok=True)
    
    print(f"Media Agent: sozdayu AI-fotosessiyu dlya '{task[:50]}...'")
    
    # Generiruyem prompty dlya fotosessii
    prompt = f"""Ty - fotograf i art-direktor.
Sozday plan AI-fotosessii (5 foto) dlya: {task}

Kazhdaye foto dolzhna byt unikalnoy, s raznym uglom, svetom, kompozitsiey.
Opishi kazhduyu v 1-2 predlozheniyakh na angliyskom (dlya AI-generatsii).

Otvet v formate JSON:
{{
  "session_name": "...",
  "style": "...",
  "photos": [
    {{"id": 1, "prompt": "detalnyy prompt na angliyskom", "description": "opisaniye na russkom"}},
    ...
  ]
}}"""

    messages = [{"role": "user", "content": prompt}]
    print("   Sozdayu plan fotosessii...")
    answer = ask_llm(messages, model="qwen2.5-coder:7b")
    
    try:
        cleaned = answer.strip()
        if cleaned.startswith("```json"): cleaned = cleaned[7:]
        if cleaned.startswith("```"): cleaned = cleaned[3:]
        if cleaned.endswith("```"): cleaned = cleaned[:-3]
        cleaned = cleaned.strip()
        session = json.loads(cleaned)
    except:
        print("   [!] Ne udalos rasparsit, ispolzuyu bazovyy plan")
        session = {
            "session_name": "Photo Session",
            "style": "Professional",
            "photos": [{"id": i, "prompt": f"professional photo, {task}", "description": f"Foto {i}"} for i in range(1, count+1)]
        }
    
    # Generiruyem foto
    photos_data = []
    for photo in session.get("photos", [])[:count]:
        fname = f"photo_{photo['id']:02d}.jpg"
        out = img_dir / fname
        try:
            generate_image(photo["prompt"], out)
            photos_data.append({"file": f"photos/{fname}", "desc": photo.get("description", "")})
        except Exception as e:
            print(f"   [!] Oshibka foto {photo['id']}: {e}")
    
    # Sozdayem HTML-albom
    photos_html = "".join([
        f'<div class="photo"><img src="{p["file"]}" alt="{p["desc"]}"><p>{p["desc"]}</p></div>'
        for p in photos_data
    ])
    
    album_html = f'''<!DOCTYPE html>
<html lang="ru">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>{session.get("session_name", "AI Photo Session")}</title>
    <script src="https://cdn.tailwindcss.com"></script>
    <style>
        body {{ background: #0a0a0a; color: white; }}
        .gallery {{ display: grid; grid-template-columns: repeat(auto-fit, minmax(300px, 1fr)); gap: 20px; padding: 40px; }}

        .photo img {{ width: 100%; height: 400px; object-fit: cover; transition: transform 0.5s; }}
        .photo:hover img {{ transform: scale(1.05); }}
        .photo p {{ position: absolute; bottom: 0; left: 0; right: 0; padding: 20px; background: linear-gradient(transparent, rgba(0,0,0,0.8)); margin: 0; }}

</head>
<body>
    <div class="text-center py-12">

"""Atlas Digital Studio - glavnyy orkestrator."""
import sys
import subprocess
import json
from pathlib import Path

def run_subprocess(script_path, args):
    result = subprocess.run(
        [sys.executable, script_path] + args,
        capture_output=True, text=False
    )
    stdout = result.stdout.decode("utf-8", errors="replace") if result.stdout else ""
    stderr = result.stderr.decode("utf-8", errors="replace") if result.stderr else ""
    print(stdout)
    if stderr:
        print(stderr)
    return result.returncode == 0

def main():
    if len(sys.argv) < 2:
        print("Atlas Digital Studio")
        print('Ispolzovaniye: python main.py "sozday lending dlya kofeyni"')
        print('              python main.py "sozday brend dlya barbershopa"')
        print('              python main.py "ai fotosessiya dlya restorana"')
        print('              python main.py "igra kliker dlya sayta"')
        sys.exit(1)

    task = sys.argv[1]
    words = task.replace('"', '').replace("'", "").split()[:3]
    project_name = "_".join(words).lower() + "_project"
    project_dir = f"./Projects/{project_name}"

    print("=" * 60)
    print("ATLAS DIGITAL STUDIO")
    print("=" * 60)
    print(f"\nZadacha: {task}\n")

    # Shag 1: Product Router
    print("--- Product Router ---")
    router_result = None
    if run_subprocess("Agents/product_router.py", [task]):
        # Chitaem rezultat iz stdout... uproshchenno
        pass
    else:
        print("[!] Router ne otvetil, ispolzuyu web po umolchaniyu")

    # Opredelyayem tip (uproshchenno - po klyuchevym slovam)
    task_lower = task.lower()
    if any(w in task_lower for w in ["brend", "brand", "logotip", "logo", "stil", "firma"]):
        product_type = "branding"
    elif any(w in task_lower for w in ["foto", "photo", "fotosessiya", "avatar", "kreativ", "media"]):
        product_type = "media"
    elif any(w in task_lower for w in ["igra", "game", "play", "platformer", "kliker"]):
        product_type = "game"
    elif any(w in task_lower for w in ["prilozheniye", "app", "programma", "utilita", "bot"]):
        product_type = "app"
    else:
        product_type = "web"

    print(f"\nVybrannyy tip produkta: {product_type.upper()}\n")

    # Shag 2: Zapuskayem nuzhnykh agentov
    if product_type == "branding":
        print("--- Branding Agent ---")
        if not run_subprocess("Agents/branding_agent.py", [task, project_dir]):
            print("\n❌ Branding Agent ne spravilsya")
            sys.exit(1)
            
    elif product_type == "media":
        print("--- Media Agent ---")
        if not run_subprocess("Agents/media_agent.py", [task, project_dir]):
            print("\n❌ Media Agent ne spravilsya")
            sys.exit(1)
            
    elif product_type == "web":
        # Klassicheskiy tsikl
        print("--- Executive ---")
        run_subprocess("Agents/executive.py", [task])
        
        print("\n--- Brief Agent ---")
        run_subprocess("Agents/brief.py", [task, project_dir])
        
        print("\n--- Image Generator ---")
        tasks_file = Path(project_dir) / "image_tasks.json"
        if tasks_file.exists():
            tasks = json.loads(tasks_file.read_text(encoding="utf-8"))
            if tasks:
                run_subprocess("Agents/image_generator.py", [json.dumps(tasks), project_dir])
        
        print("\n--- Web Developer ---")
        skill = sys.argv[2] if len(sys.argv) > 2 else "product_showcase"
        if not run_subprocess("Agents/developer.py", [task, project_dir, skill]):
            print("\n❌ Developer ne spravilsya")
            sys.exit(1)
    
    else:
        print(f"[INFO] Tip '{product_type}' v razrabotke. Ispolzuyu web kak fallback.")
        # Fallback na web
        skill = sys.argv[2] if len(sys.argv) > 2 else "product_showcase"
        run_subprocess("Agents/developer.py", [task, project_dir, skill])

    print("\n" + "=" * 60)
    print("✅ ATLAS ZAVERSHIL RABOTU")
    print("=" * 60)
    print(f"\n📁 Proekt: {Path(project_dir).absolute()}")

if __name__ == "__main__":
    main()
